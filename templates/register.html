@app.route("/register", methods=["GET", "POST"]) def register(): """Register user""" # Forget any user_id session.clear() if request.method == "POST": username = request.form.get("username") password = request.form.get("password") confirmation = request.form.get("confirmation") # 1. Ensure all fields are filled if not username or not password or not confirmation: return apology("all fields required", 400) # 2. Ensure passwords match if password != confirmation: return apology("passwords must match", 400) # 3. Hash password and insert hash_pw = generate_password_hash(password) try: # db.execute returns the ID of the new row user_id = db.execute("INSERT INTO users (username, hash) VALUES (?, ?)", username, hash_pw) except: # Triggers if username exists due to UNIQUE index return apology("username taken", 400) # 4. Remember which user has logged in (Crucial for "success" check) session["user_id"] = user_id # 5. Redirect user to home page flash("Registered!") return redirect("/") else: return render_template("register.html")
